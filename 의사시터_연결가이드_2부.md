## 🔧 Step 1: VetService 파일 생성

**파일 경로:** `lib/services/vet_service.dart`

**전체 코드를 복사해서 붙여넣으세요:**

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;

class VetService {
  // Android 에뮬레이터용 baseURL
  static const String baseUrl = 'http://10.0.2.2:9075/api/v1';

  // 이미지 URL을 완전한 URL로 변환
  static String getFullImageUrl(String? imageUrl) {
    if (imageUrl == null || imageUrl.isEmpty) return '';
    if (imageUrl.startsWith('http://') || imageUrl.startsWith('https://')) {
      return imageUrl;
    }
    return 'http://10.0.2.2:9075$imageUrl';
  }

  /// 파트너(수의사/펫시터) 목록 조회
  static Future<Map<String, dynamic>> getPartners(String userType) async {
    try {
      final response = await http.get(
        Uri.parse('$baseUrl/users/partners?user_type=$userType'),
        headers: {'Content-Type': 'application/json'},
      );

      if (response.statusCode == 200) {
        final jsonResponse = json.decode(response.body);
        return {
          'success': jsonResponse['success'] ?? true,
          'data': jsonResponse['data'],
        };
      } else {
        throw Exception('파트너 목록을 불러오는데 실패했습니다');
      }
    } catch (e) {
      print('파트너 조회 오류: $e');
      return {
        'success': false,
        'message': e.toString(),
      };
    }
  }

  /// 예약 생성
  static Future<Map<String, dynamic>> createReservation({
    required int userId,
    required int partnerId,
    required String serviceCategorical,
    required int petId,
    String? content,
    String? vetSpecialty,
    String? petsitterWork,
  }) async {
    try {
      final Map<String, dynamic> requestBody = {
        'userId': userId,
        'partnerId': partnerId,
        'serviceCategorical': serviceCategorical,
        'petId': petId,
      };

      if (content != null && content.isNotEmpty) {
        requestBody['reservationContent'] = content;
      }
      if (vetSpecialty != null && vetSpecialty.isNotEmpty) {
        requestBody['vetSpecialtyCsv'] = vetSpecialty;
      }
      if (petsitterWork != null && petsitterWork.isNotEmpty) {
        requestBody['petsitterWorkCsv'] = petsitterWork;
      }

      final response = await http.post(
        Uri.parse('$baseUrl/reservations'),
        headers: {'Content-Type': 'application/json'},
        body: json.encode(requestBody),
      );

      if (response.statusCode == 200) {
        final jsonResponse = json.decode(response.body);
        return {
          'success': jsonResponse['success'] ?? true,
          'data': jsonResponse['data'],
        };
      } else {
        throw Exception('예약 생성에 실패했습니다');
      }
    } catch (e) {
      print('예약 생성 오류: $e');
      return {
        'success': false,
        'message': e.toString(),
      };
    }
  }

  /// 내 예약 목록 조회
  static Future<Map<String, dynamic>> getMyReservations({
    required int userId,
    String? serviceType,
  }) async {
    try {
      String url = '$baseUrl/reservations/mine?userId=$userId';
      if (serviceType != null && serviceType.isNotEmpty) {
        url += '&serviceType=$serviceType';
      }

      final response = await http.get(
        Uri.parse(url),
        headers: {'Content-Type': 'application/json'},
      );

      if (response.statusCode == 200) {
        final jsonResponse = json.decode(response.body);
        return {
          'success': jsonResponse['success'] ?? true,
          'data': jsonResponse['data'],
        };
      } else {
        throw Exception('예약 목록을 불러오는데 실패했습니다');
      }
    } catch (e) {
      print('예약 목록 조회 오류: $e');
      return {
        'success': false,
        'message': e.toString(),
      };
    }
  }
}
```

```dart
Future<void> loadVets() async {
  _isLoadingVets = true;
  _vetsError = null;
  notifyListeners();

  try {
    // 백엔드 API 호출
    final response = await VetService.getPartners('HOSPITAL');

    if (response['success'] == true) {
      final data = response['data'];
      final List<dynamic> result = data['result'] ?? [];

      // 백엔드 응답을 VetModel로 변환
      _vets = result.map((partnerJson) {
        return VetModel(
          id: partnerJson['id'].toString(),
          name: partnerJson['username'] ?? '',
          doctorName: partnerJson['username'] ?? '',
          address: '주소 정보 없음',
          phone: '전화번호 없음',
          rating: 4.5,
          specialties: _parseSpecialties(partnerJson['vet_specialty']),
          availableTimes: _parseWorkingHours(partnerJson['working_schedule']),
          distance: 0.0,
          isOpen: true,
          imageUrl: VetService.getFullImageUrl(partnerJson['image_url']),
          description: '수의사 소개',
        );
      }).toList();
    } else {
      _vetsError = response['message'] ?? '수의사 목록을 불러올 수 없습니다';
      _vets = [];
    }
  } catch (e) {
    _vetsError = e.toString();
    _vets = [];
  } finally {
    _isLoadingVets = false;
    notifyListeners();
  }
}
```

```dart
// 진료과목 파싱 (백엔드 enum을 한글로 변환)
List<String> _parseSpecialties(dynamic specialties) {
  if (specialties == null) return [];

  final Map<String, String> specialtyMap = {
    'INTERNAL_MEDICINE': '내과',
    'SURGERY': '외과',
    'ORTHOPEDICS': '정형외과',
    'OPHTHALMOLOGY': '안과',
    'DENTISTRY': '치과',
    'DERMATOLOGY': '피부과',
    'EMERGENCY_MEDICINE': '응급의학과',
    'GENERAL': '일반',
  };

  if (specialties is List) {
    return specialties
        .map((s) => specialtyMap[s] ?? s.toString())
        .toList();
  }
  return [];
}

// 근무시간 파싱
List<String> _parseWorkingHours(dynamic schedule) {
  if (schedule == null) return [];

  try {
    String start = schedule['working_start_hours'] ?? '09:00';
    String end = schedule['working_end_hours'] ?? '18:00';

    List<String> times = [];
    int startHour = int.parse(start.split(':')[0]);
    int endHour = int.parse(end.split(':')[0]);

    for (int hour = startHour; hour < endHour; hour++) {
      times.add('${hour.toString().padLeft(2, '0')}:00');
    }

    return times;
  } catch (e) {
    return ['09:00', '10:00', '11:00', '14:00', '15:00', '16:00'];
  }
}

// 펫시터 업무 파싱
List<String> _parsePetsitterWork(dynamic works) {
  if (works == null) return [];

  final Map<String, String> workMap = {
    'WALK': '산책',
    'TRANSPORT': '이동',
    'HYGIENE': '위생관리',
    'ALL': '전체',
  };

  if (works is List) {
    return works.map((w) => workMap[w] ?? w.toString()).toList();
  }
  return [];
}
```

```dart
/// 예약 목록 로드
Future<void> loadReservations() async {
  _isLoadingReservations = true;
  _reservationsError = null;
  notifyListeners();

  try {
    final currentUser = AuthService.getCurrentUser();
    if (currentUser == null) {
      _reservationsError = '로그인이 필요합니다';
      _reservations = [];
      _isLoadingReservations = false;
      notifyListeners();
      return;
    }

    final response = await VetService.getMyReservations(
      userId: currentUser.id,
      serviceType: 'HOSPITAL',
    );

    if (response['success'] == true) {
      final List<dynamic> data = response['data'] ?? [];

      _reservations = data.map((json) {
        return ReservationModel(
          id: json['reservationId'].toString(),
          vetId: json['partnerId'].toString(),
          vetName: json['partnerName'] ?? '',
          petName: json['petName'] ?? '',
          petType: '강아지',
          specialty: json['serviceCategorical'] ?? '',
          reservationDate: DateTime.parse(json['createdAt']),
          timeSlot: '시간 미정',
          status: _parseStatus(json['status']),
          memo: json['reservationContent'] ?? '',
        );
      }).toList();
    } else {
      _reservationsError = response['message'] ?? '예약 목록을 불러올 수 없습니다';
      _reservations = [];
    }
  } catch (e) {
    _reservationsError = e.toString();
    _reservations = [];
  } finally {
    _isLoadingReservations = false;
    notifyListeners();
  }
}

// 예약 상태 파싱
String _parseStatus(String? status) {
  final Map<String, String> statusMap = {
    'WAITING': '대기중',
    'CONFIRMED': '승인됨',
    'CHECKED_IN': '체크인 완료',
    'COMPLETED': '완료',
    'CANCELLED_BY_USER': '사용자 취소',
    'CANCELLED_BY_BIZ': '업체 취소',
  };

  return statusMap[status] ?? '알 수 없음';
}
```

## 🐞 문제 해결

### ❌ "SocketException: Failed to connect"
- 백엔드 서버 실행 확인: `./gradlew bootRun`
- Android 에뮬레이터에서는 `10.0.2.2` 사용

### ❌ 빈 목록이 표시됨
- 데이터베이스에 수의사 데이터가 없을 수 있음
- Postman으로 수의사 유저 생성 필요

### ❌ 이미지가 표시되지 않음
- `VetService.getFullImageUrl()` 사용 확인
- 백엔드에서 이미지 업로드 완료 확인
